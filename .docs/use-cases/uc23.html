Implemente esta funcionalidade completa no banco, na API e na interface React. Entregue somente esta atividade, com seus testes, em uma branch uc23 do seu fork pessoal.

Pré-requisitos e momento de começar

  • UC18 — Cancelar reserva de exemplar
  • UC20 — Registrar devolução
Aguarde a integração pelo professor. Um PR aberto ou aprovado ainda não basta: os pré-requisitos precisam estar na branch central saas-evolution/fase01. Nunca comece a partir da branch de outro aluno. O comando de preparação abaixo confere o código integrado no remoto central antes de criar sua branch.

Ambiente: Git, Bun 1.4 ou mais novo e Docker Desktop em execução. Os comandos de terminal usam shell compatível com Bash; no Windows, use Git Bash. Use os lockfiles existentes e o PostgreSQL do Compose.

Contrato e critérios de aceite

  • O catálogo contém exatamente 50 autores reais, entre um e cinco livros reais por autor e ISBN-13 válido, com links de fontes bibliográficas versionados.
  • Cada livro do catálogo recebe entre um e dez exemplares físicos, respeitando o contador de registros e os limites de autores.
  • O seed cria 120 leitores fictícios, com nome, CPF válido e único, e-mail e telefone obrigatórios.
  • A fotografia inicial contém 120 empréstimos devolvidos, 30 não devolvidos no prazo e 30 não devolvidos atrasados: 180 empréstimos no total.
  • A fotografia inicial contém 20 reservas ativas, 10 canceladas e 10 expiradas, com a janela de sete dias corridos e datas civis de São Paulo.
  • Nenhum leitor excede três empréstimos não devolvidos ou duas reservas ativas; exemplares com empréstimo aberto não possuem reserva ativa.
  • A carga é explícita, transacional e idempotente; uma repetição ou requisição concorrente não duplica os registros e preserva dados existentes.
  • GET /api/seed informa a execução e o resumo; POST /api/seed executa a carga; a tela /livros/seed e o CLI oferecem a mesma operação.
  • O relatório mantém a data de referência da execução e não restaura registros alterados manualmente posteriormente.

Respostas de erro usam {"error":"mensagem"}: 400 para entrada inválida, 404 para registro ausente e 409 para conflito de regra ou vínculo. Cadastros retornam 201; atualizações e ações retornam 200; exclusões retornam 204.

A transação PostgreSQL usa um advisory lock compartilhado. A consulta de disponibilidade/limites e sua alteração ocorrem na mesma transação, incluindo requisições simultâneas de processos diferentes. Os testes legados recebem limpeza restrita a bancos cujo nome termina em _test.

export function getBodyAsObject(body: unknown): Record<string, unknown> {
  if (typeof body !== "object" || body === null) {
    throw new InvalidInput("o corpo da requisição deve ser um objeto JSON");
  }

  return body as Record<string, unknown>;
}

export function getFieldAsText(body: Record<string, unknown>, field: string): string {
  const value = body[field];

  if (typeof value !== "string" || value.trim() === "") {
    throw new InvalidInput(`campo obrigatório: ${field}`);
  }

  return value.trim();
}

export function getFieldAsPositiveInt(
  body: Record<string, unknown>,
  field: string,
): number {
  const raw = body[field];
  if ((typeof raw !== "number" && typeof raw !== "string") ||
      (typeof raw === "string" && !/^\d+$/.test(raw))) {
    throw new InvalidInput(`campo inválido: ${field}`);
  }
  const value = Number(raw);

  if (!Number.isSafeInteger(value) || value <= 0) {
    throw new InvalidInput(`campo inválido: ${field}`);
  }

  return value;
}

import type { TransactionSQL } from "bun";
import { sql } from "./postgres";

/**
 * Serializa as escritas desta versão didática na mesma transação PostgreSQL.
 * A verificação de um limite e sua alteração nunca ficam separadas por COMMIT.
 * O lock é compartilhado entre processos; um mutex JavaScript não bastaria.
 */
export async function emTransacao<T>(
  executar: (tx: TransactionSQL) => Promise<T>,
): Promise<T> {
  return await sql.begin(async (tx) => {
    await tx`SELECT pg_advisory_xact_lock(22001)`;
    return await executar(tx);
  }) as T;
}

export async function carregarCasosDeUso(): Promise<CasoDeUso[]> {
  const arquivos = Array.from(
    new Glob("uc[0-9][0-9].ts").scanSync({ cwd: import.meta.dir }),
  ).sort();
  const casos: CasoDeUso[] = [];
  for (const arquivo of arquivos) {
    const modulo = await import(new URL(arquivo, import.meta.url).href);
    const caso = modulo.default as CasoDeUso;
    if (!caso || caso.id !== arquivo.slice(0, 4) || typeof caso.registrar !== "function") {
      throw new Error(`Caso de uso inválido: ${arquivo}`);
    }
    casos.push(caso);
  }
  return casos;

/** O build do React (`bun run build` em frontend/), relativo à pasta backend/. */
const FRONTEND = "../frontend/dist";

function errorResponse(error: unknown): Response {
  if (error instanceof InvalidInput || error instanceof InvalidValue || error instanceof SyntaxError) {
    return Response.json({ error: error.message }, { status: 400 });
  }

  if (error instanceof NotFound) {
    return Response.json({ error: error.message }, { status: 404 });
  }

  if (error instanceof RuleConflict || error instanceof DomainError) {
    return Response.json({ error: error.message }, { status: 409 });
  }

  console.error(error);

  return Response.json({ error: "Erro interno" }, { status: 500 });
}

export function createApp(useCases: UseCases, casos: CasoDeUso[] = []) {
  const api = new Hono();

  api.get("/", (contexto) => contexto.json({ message: "API da Biblioteca" }));

  // Mantém os dois casos originais até suas atividades de evolução serem integradas.
  const ids = new Set(casos.map((caso) => caso.id));
  if (!ids.has("uc01")) registerCadastrarLegado(api, useCases);
  if (!ids.has("uc01") && !ids.has("uc02")) registerBuscarLegado(api, useCases);
  for (const caso of casos) caso.registrar(api);

  const app = new Hono();

  // toda a API mora sob /api: o resto dos endereços fica livre para o frontend
  app.route("/api", api);
  app.all("/api/*", (contexto) => contexto.notFound());

  // arquivos do build: index.html, JS, CSS, ícone
  app.use("*", serveStatic({ root: FRONTEND }));

  // qualquer outro endereço (/cadastrar, por exemplo) é rota do React Router:
  // devolve o index.html e o roteador do navegador decide o que mostrar
  app.get("*", serveStatic({ path: `${FRONTEND}/index.html` }));

  app.notFound((contexto) =>
    contexto.json({ error: "Recurso não encontrado" }, 404),
  );

  app.onError((error) => errorResponse(error));

  return app;
}

export function createServer(
  useCases: UseCases,
  porta = Number(process.env.PORT ?? 3000),
  casos: CasoDeUso[] = [],
) {
  return Bun.serve({ port: porta, fetch: createApp(useCases, casos).fetch });
}

import { Glob } from "bun";
import { dirname, join, normalize } from "node:path";

const problems: string[] = [];

/** Quem é o dono de cada tabela. Nenhum módulo cita uma tabela que não é dele. */
const TABELAS: Record<string, string> = {
  livros: "acervo",
  autores: "autoria",
};

for await (const rawFile of new Glob("src/modules/**/*.ts").scan(".")) {
  const file = rawFile.replaceAll("\\", "/");
  const module = file.split("/")[2]!;
  const code = await Bun.file(file).text();

  for (const found of code.matchAll(/from\s+"(\.[^"]+)"/g)) {
    const target = normalize(join(dirname(file), found[1]!)).replaceAll("\\", "/");

    if (!target.startsWith("src/modules/")) continue;

    const otherModule = target.split("/")[2]!;
    if (otherModule === module) continue;

    if (target !== `src/modules/${otherModule}`) {
      problems.push(`${file}\n      importa ${target}`);
    }
  }

  for (const [tabela, dono] of Object.entries(TABELAS)) {
    if (dono === module) continue;

    if (new RegExp(`\\b(FROM|JOIN|INTO|UPDATE|TABLE|EXISTS)\\s+${tabela}\\b`, "i").test(code)) {
      problems.push(
        `${file}\n      toca a tabela "${tabela}", que pertence a ${dono}`,
      );
    }
  }
}

if (problems.length > 0) {
  console.error("Importações que furam a fronteira de módulo:\n");
  for (const problema of problems) console.error(`  ${problema}\n`);
  process.exit(1);
}

console.log("Fronteiras de módulo respeitadas.");

import { emTransacao } from "../src/infrastructure/transaction";

/** Só a suíte legada precisa zerar o acervo; jamais permita executar isso no banco da aula. */
export async function limparAcervoDeTeste(): Promise<void> {
  const url = new URL(process.env.DATABASE_URL ?? "postgres://sem-banco/nao-configurado");
  if (!decodeURIComponent(url.pathname).endsWith("_test")) {
    throw new Error("A limpeza dos testes exige DATABASE_URL apontando para um banco com nome terminado em _test.");
  }
  await emTransacao(async tx => {
    // A aplicação pode estar em uma etapa anterior: as tabelas novas ainda podem não existir.
    for (const tabela of ["reservas", "emprestimos", "exemplares"] as const) {
      const [row] = await tx`SELECT to_regclass(${tabela}) AS tabela`;
      if (row?.tabela) await tx.unsafe(`DELETE FROM ${tabela}`);
    }
    await tx`DELETE FROM livros`;
  });
}

export async function api<T>(caminho: string, opcoes: RequestInit = {}): Promise<T> {
  let resposta: Response;
  try {
    resposta = await fetch(caminho, opcoes);
  } catch {
    throw new Error("Não foi possível conectar à biblioteca. Verifique a conexão e tente novamente.");
  }
  if (resposta.status === 204) return undefined as T;
  const corpo = await resposta.json().catch(() => ({}));
  if (!resposta.ok) {
    throw new Error(typeof corpo.error === "string" ? corpo.error : `Não foi possível concluir a operação (HTTP ${resposta.status}).`);
  }
  return corpo as T;
}

import { useEffect, useState, type FormEvent } from "react";
import { useSearchParams } from "react-router-dom";
import { buscarLivros, formatarData, type Livro } from "../api.ts";

/** A resposta da API, junto com o termo que a produziu. */
type Resultado = {
  termo: string;
  livros: Livro[];
  erro: string;
};

export default function BuscarLivros() {
  // o termo mora na URL (/?q=...): recarregar a página repete a busca
  const [params, setParams] = useSearchParams();
  const termo = params.get("q") ?? "";

  const [resultado, setResultado] = useState<Resultado | null>(null);

  useEffect(() => {
    if (termo === "") return;

    // se o termo mudar antes da resposta chegar, a resposta antiga é ignorada
    let ignorar = false;

    buscarLivros(termo)
      .then((livros) => {
        if (!ignorar) setResultado({ termo, livros, erro: "" });
      })
      .catch((e: Error) => {
        if (!ignorar) setResultado({ termo, livros: [], erro: e.message });
      });

    return () => {
      ignorar = true;
    };
  }, [termo]);

  // ainda não chegou resposta para o termo que está na URL
  const carregando = resultado?.termo !== termo;

  function buscar(evento: FormEvent<HTMLFormElement>) {
    evento.preventDefault();

    const q = String(new FormData(evento.currentTarget).get("q")).trim();

    setParams(q === "" ? {} : { q });
  }

  return (
    <section>
      {/* React 19: <title> dentro do componente vai parar no <head> */}
      <title>Buscar livros · Biblioteca</title>

      <h1>Buscar livros</h1>

      <form className="busca" onSubmit={buscar}>
        <input
          // key: quando a URL muda (menu, voltar), o campo recomeça com o termo dela
          key={termo}
          name="q"
          defaultValue={termo}
          placeholder="ISBN, título ou nome do autor"
          aria-label="Termo de busca"
        />
        <button type="submit">Buscar</button>
      </form>

      {termo === "" ? (
        <p className="dica">Digite um ISBN, parte do título ou o nome do autor.</p>
      ) : carregando ? (
        <p className="dica">Buscando…</p>
      ) : resultado.erro !== "" ? (
        <p className="erro">{resultado.erro}</p>
      ) : resultado.livros.length === 0 ? (
        <p className="dica">Nenhum livro encontrado para “{termo}”.</p>
      ) : (
        <div
          className="resultado-busca"
          role="region"
          aria-label="Livros encontrados"
          tabIndex={0}
        >
          <table>
            <thead>
              <tr>
                <th>ISBN</th>
                <th>Título</th>
                <th>Autor</th>
                <th>Catalogado em</th>
              </tr>
            </thead>
            <tbody>
              {resultado.livros.map((livro) => (
                <tr key={livro.id}>
                  <td>{livro.isbn}</td>
                  <td>{livro.titulo}</td>
                  <td>{livro.autor}</td>
                  <td>{formatarData(livro.dataCatalogacao)}</td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      )}
    </section>
  );
}

import { useState, type FormEvent } from "react";
import { Link } from "react-router-dom";
import { cadastrarLivro, formatarData, type Livro } from "../api.ts";

export default function CadastrarLivro() {
  const [salvo, setSalvo] = useState<Livro | null>(null);
  const [erro, setErro] = useState("");
  const [enviando, setEnviando] = useState(false);

  async function cadastrar(evento: FormEvent<HTMLFormElement>) {
    evento.preventDefault();

    const formulario = evento.currentTarget;
    const dados = new FormData(formulario);

    setEnviando(true);
    setErro("");
    setSalvo(null);

    try {
      const livro = await cadastrarLivro({
        isbn: String(dados.get("isbn")),
        titulo: String(dados.get("titulo")),
        autorId: Number(dados.get("autorId")),
      });

      setSalvo(livro);
      formulario.reset();
    } catch (e) {
      // as regras continuam no backend: aqui só mostramos o que ele respondeu
      setErro((e as Error).message);
    } finally {
      setEnviando(false);
    }
  }

  return (
    <section>
      <title>Cadastrar livro · Biblioteca</title>

      <h1>Cadastrar livro</h1>

      <form className="cadastro" onSubmit={cadastrar}>
        <label>
          ISBN
          <input name="isbn" required placeholder="9780321125217" />
        </label>

        <label>
          Título
          <input name="titulo" required />
        </label>

        <label>
          Autor (id)
          <input name="autorId" type="number" min="1" required />
          <small>
            Carga inicial: 1 Eric Evans · 2 Robert C. Martin · 3 Harper Lee · 4 Jane
            Austen
          </small>
        </label>

        <button type="submit" disabled={enviando}>
          {enviando ? "Cadastrando…" : "Cadastrar"}
        </button>
      </form>

      {erro !== "" && <p className="erro">{erro}</p>}

      {salvo !== null && (
        <div className="sucesso">
          <p>
            <strong>{salvo.titulo}</strong> entrou no catálogo em {formatarData(salvo.dataCatalogacao)}.
          </p>
          <Link to={`/?q=${salvo.isbn}`}>Ver no acervo</Link>
        </div>
      )}
    </section>
  );
}

import { Isbn } from "../modules/acervo/domain/Isbn";
import { RuleConflict } from "../shared/errors";

export const VERSAO_SEED = "uc23-v1";
export type CatalogoSeed = { fonte: string; consultadoEm: string; autores: { nome: string; fonte: string; livros: { titulo: string; isbn: string; fonte: string }[] }[] };
export type ResumoSeed = { autores: number; livros: number; exemplares: number; leitores: number; emprestimosDevolvidos: number; emprestimosEmDia: number; emprestimosAtrasados: number; reservasAtivas: number; reservasCanceladas: number; reservasExpiradas: number };
export type RelatorioSeed = { executado: boolean; resumo?: ResumoSeed; dataReferencia?: string };

export function validarCatalogo(catalogo: CatalogoSeed): void {
  if (!Array.isArray(catalogo.autores) || catalogo.autores.length !== 50) throw new RuleConflict("O catálogo do seed deve ter exatamente 50 autores reais.");
  const autores = new Set<string>(); const isbns = new Set<string>();
  for (const autor of catalogo.autores) {
    const nome = autor.nome.trim().toLowerCase();
    if (!nome || !autor.fonte || autores.has(nome) || autor.livros.length < 1 || autor.livros.length > 5) throw new RuleConflict("Catálogo do seed inválido: autor repetido, sem fonte ou fora do limite de 1 a 5 livros.");
    autores.add(nome);
    for (const livro of autor.livros) {
      if (!livro.titulo.trim() || !livro.fonte || !Isbn.isValid(livro.isbn) || isbns.has(livro.isbn)) throw new RuleConflict(`Catálogo do seed inválido: título, fonte ou ISBN ${livro.isbn}.`);
      isbns.add(livro.isbn);
    }
  }
}

export function cpfFicticio(indice: number): string {
  let cpf = `900${String(indice).padStart(6, "0")}`;
  for (const tamanho of [9, 10]) {
    const soma = [...cpf].reduce((total, n, i) => total + Number(n) * (tamanho + 1 - i), 0);
    const resto = soma * 10 % 11; cpf += String(resto === 10 ? 0 : resto);
  }
  return cpf;
}

import { sql } from "../infrastructure/postgres";
import { hoje, somarDias } from "../modules/circulacao/domain/modelo";

type Copia = { id: number; data_catalogacao: string };
async function marcar(tx: TransactionSQL, entidade: string, id: number) {
  await tx`INSERT INTO seed_registros (versao, entidade, registro_id) VALUES (${VERSAO_SEED}, ${entidade}, ${id}) ON CONFLICT DO NOTHING`;

/** Orquestração de integração fora de modules: todas as escritas são uma transação única. */
export async function executarSeed(db: SQL = sql, agora = new Date(), fornecido?: CatalogoSeed) {
  const catalogo = fornecido ?? await Bun.file(new URL("./catalogo.json", import.meta.url)).json() as CatalogoSeed;
  validarCatalogo(catalogo);
  await prepararSeed(db);
  return await db.begin(async tx => {
    await tx`SELECT pg_advisory_xact_lock(22001)`;
    const anterior = await relatorioSeed(tx);
    if (anterior.executado) return { ...anterior, criado: false };
    const data = hoje(() => agora); const antiga = somarDias(data, -180);
    const todos: Copia[] = []; let quantidadeLivros = 0;
    for (const autor of catalogo.autores) {
      const existentes = await tx`SELECT id FROM autores WHERE lower(btrim(nome)) = lower(btrim(${autor.nome}))`;
      if (existentes.length > 1) throw new RuleConflict(`Seed cancelado: há mais de um autor chamado ${autor.nome}.`);
      const [novo] = existentes.length ? existentes : await tx`INSERT INTO autores (nome, tipo) VALUES (${autor.nome}, 'literatura') RETURNING id`;
      const autorId = Number(novo!.id); await marcar(tx, "autores", autorId);
      const atuais = await tx<{ id: number; isbn: string }[]>`SELECT id, isbn FROM livros WHERE autor_id = ${autorId}`;
      const adicionais = autor.livros.filter(l => !atuais.some(r => r.isbn === l.isbn)).length;
      if (atuais.length + adicionais > 5) throw new RuleConflict(`Seed cancelado: ${autor.nome} ultrapassaria 5 livros. Nenhum dado foi alterado.`);
      for (const livro of autor.livros) {
        const [atual] = await tx`SELECT id, titulo, autor_id, data_catalogacao FROM livros WHERE isbn = ${livro.isbn}`;
        if (atual && (atual.autor_id !== autorId || atual.titulo.trim().toLowerCase() !== livro.titulo.trim().toLowerCase())) throw new RuleConflict(`Seed cancelado: ISBN ${livro.isbn} já pertence a outro registro bibliográfico.`);
        if (!atual) {
          const [titulo] = await tx`SELECT id FROM livros WHERE autor_id = ${autorId} AND lower(btrim(titulo)) = lower(btrim(${livro.titulo}))`;
          if (titulo) throw new RuleConflict(`Seed cancelado: título ${livro.titulo} já existe com outro ISBN.`);
        }
        const [inserido] = atual ? [atual] : await tx`INSERT INTO livros (isbn, titulo, autor_id, data_catalogacao) VALUES (${livro.isbn}, ${livro.titulo}, ${autorId}, ${antiga}) RETURNING id, data_catalogacao`;
        const livroId = Number(inserido!.id); await marcar(tx, "livros", livroId);
        const copias = await tx<Copia[]>`SELECT id, data_catalogacao FROM exemplares WHERE livro_id = ${livroId} ORDER BY id`;
        if (copias.length > 10) throw new RuleConflict(`Seed cancelado: o livro ${livro.titulo} já possui mais de 10 exemplares.`);
        const desejadas = 1 + quantidadeLivros++ % 10;
        const dataCopia = String(inserido!.data_catalogacao) > antiga ? String(inserido!.data_catalogacao) : antiga;
        for (let i = copias.length; i < desejadas; i++) {
          const ano = dataCopia.slice(0, 4);
          const [contador] = await tx`INSERT INTO acervo_sequencias (ano, ultimo) VALUES (${ano}, 1) ON CONFLICT (ano) DO UPDATE SET ultimo = acervo_sequencias.ultimo + 1 RETURNING ultimo`;
          const registro = `${ano}-${String(contador!.ultimo).padStart(6, "0")}`;
          const [copia] = await tx<Copia[]>`INSERT INTO exemplares (livro_id, numero_registro, localizacao, estado_conservacao, data_catalogacao) VALUES (${livroId}, ${registro}, 'Acervo demonstrativo UC23', 'bom', ${dataCopia}) RETURNING id, data_catalogacao`;
          copias.push(copia!);
        }
        for (const copia of copias) {
          if (copia.data_catalogacao > data || copia.data_catalogacao < String(inserido!.data_catalogacao)) throw new RuleConflict("Seed cancelado: há datas de catalogação incompatíveis na base existente.");
          await marcar(tx, "exemplares", copia.id); todos.push(copia);
        }
      }
      await tx`UPDATE autores SET livros_no_acervo = (SELECT COUNT(*) FROM livros WHERE autor_id = ${autorId}) WHERE id = ${autorId}`;
    }
    const livres: Copia[] = [];
    for (const copia of todos) {
      // Movimentações de demonstração só usam cópias antigas e sem histórico anterior.
      // Assim não alteramos nem sobrepomos a circulação dos dados preexistentes.
      if (copia.data_catalogacao > antiga) continue;
      const [historico] = await tx`SELECT EXISTS(SELECT 1 FROM emprestimos WHERE exemplar_id = ${copia.id}) OR EXISTS(SELECT 1 FROM reservas WHERE exemplar_id = ${copia.id}) AS existe`;
      if (!historico!.existe) livres.push(copia);
    }
    const historicos = livres;
    if (todos.length < 100 || livres.length < 100) throw new RuleConflict("Seed cancelado: são necessários 100 exemplares catalogados há pelo menos 180 dias e sem histórico anterior. Nenhum dado foi alterado.");
    const leitores: number[] = [];
    for (let i = 1; i <= 120; i++) {
      const dados = validarDadosLeitor({ nome: `Leitor Fictício ${String(i).padStart(3, "0")}`, cpf: cpfFicticio(i), email: `leitor.seed${String(i).padStart(3, "0")}@example.com`, telefone: `2799${String(i).padStart(7, "0")}` });
      const [existente] = await tx`SELECT id, nome, email, telefone FROM leitores WHERE cpf = ${dados.cpf}`;
      if (existente && (existente.nome !== dados.nome || existente.email !== dados.email || existente.telefone !== dados.telefone)) throw new RuleConflict(`Seed cancelado: CPF fictício ${dados.cpf} já pertence a outro leitor.`);
      if (existente) {
        const [vinculado] = await tx`SELECT EXISTS(SELECT 1 FROM reservas WHERE leitor_id = ${existente.id}) OR EXISTS(SELECT 1 FROM emprestimos WHERE leitor_id = ${existente.id}) AS existe`;
        if (vinculado!.existe) throw new RuleConflict("Seed cancelado: um leitor fictício já possui histórico incompatível.");
      }
      const [salvo] = existente ? [existente] : await tx`INSERT INTO leitores (nome, cpf, email, telefone) VALUES (${dados.nome}, ${dados.cpf}, ${dados.email}, ${dados.telefone}) RETURNING id`;
      leitores.push(Number(salvo!.id)); await marcar(tx, "leitores", Number(salvo!.id));
    }
    for (let i = 0; i < 120; i++) {
      const copia = historicos[i % historicos.length]!;
      // Cada volta usa uma quinzena anterior, sem sobrepor os empréstimos da mesma cópia.
      const inicio = somarDias(data, -60 - Math.floor(i / historicos.length) * 15);
      const [r] = await tx`INSERT INTO emprestimos (leitor_id, exemplar_id, data_emprestimo, data_prevista_devolucao, data_devolucao) VALUES (${leitores[i]!}, ${copia.id}, ${inicio}, ${somarDias(inicio, 14)}, ${somarDias(inicio, 7)}) RETURNING id`;
      await marcar(tx, "emprestimos", Number(r!.id));
    }
    for (let i = 0; i < 60; i++) {
      const inicio = somarDias(data, i < 30 ? -3 : -21);
      const [r] = await tx`INSERT INTO emprestimos (leitor_id, exemplar_id, data_emprestimo, data_prevista_devolucao) VALUES (${leitores[i]!}, ${livres[i]!.id}, ${inicio}, ${somarDias(inicio, 14)}) RETURNING id`;
      await marcar(tx, "emprestimos", Number(r!.id));
    }
    for (let i = 0; i < 40; i++) {
      const inicio = somarDias(data, i < 20 ? 0 : i < 30 ? -10 : -8);
      const situacao = i >= 20 && i < 30 ? "cancelada" : "ativa";
      const [r] = await tx`INSERT INTO reservas (leitor_id, exemplar_id, data_reserva, data_limite_retirada, data_expiracao, situacao) VALUES (${leitores[i + 60]!}, ${livres[i + 60]!.id}, ${inicio}, ${somarDias(inicio, 6)}, ${somarDias(inicio, 7)}, ${situacao}) RETURNING id`;
      await marcar(tx, "reservas", Number(r!.id));
    }
    const resumo: ResumoSeed = { autores: 50, livros: quantidadeLivros, exemplares: todos.length, leitores: 120, emprestimosDevolvidos: 120, emprestimosEmDia: 30, emprestimosAtrasados: 30, reservasAtivas: 20, reservasCanceladas: 10, reservasExpiradas: 10 };
    await tx`INSERT INTO seed_execucoes (versao, data_referencia, resumo) VALUES (${VERSAO_SEED}, ${data}, ${JSON.stringify(resumo)}::text::jsonb)`;
    return { executado: true, criado: true, dataReferencia: data, resumo };
  });
}
